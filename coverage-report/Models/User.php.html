313    /**
314     * Tres niveles, de mayor a menor:
315     *
316     *   1. El superadministrador y el dueño del rancho no tienen restricción.
317     *      Quien manda no se limita a sí mismo.
318     *   2. Un `admin` dentro del rancho de otro maneja toda la operación,
319     *      pero no las acciones críticas, que van por Gate aparte.
320     *   3. Un trabajador recibe lo de su puesto, más lo que se le haya
321     *      concedido a título personal, menos lo que se le haya quitado.
322     *
323     * Una cuenta desactivada no conserva permisos: ni siquiera llega aquí,
324     * pero se deja explícito por si alguien consulta el modelo directamente.
325     */
326    private function resolverPermisos(): array
327    {
328        if ($this->activo === false) {
329            return [];
330        }
331
332        if ($this->isSuperAdmin() || $this->esDuenoDeCuenta()) {
333            return array_fill_keys(ModuloSistema::claves(), ModuloSistema::TODAS);
334        }
335
336        if ($this->esAdministradorExacto()) {
337            return array_fill_keys(ModuloSistema::claves(), ModuloSistema::TODAS);
338        }
339
340        $permisos = $this->puestoCatalogo?->permisosNormalizados() ?? [];
341
342        $extra = $this->permisos_extra ?? [];
343
344        foreach ($extra['conceder'] ?? [] as $modulo => $acciones) {
345            $permisos[$modulo] = array_values(array_unique(
346                array_merge($permisos[$modulo] ?? [], (array) $acciones)
347            ));
348        }
349
350        foreach ($extra['revocar'] ?? [] as $modulo => $acciones) {
351            $restantes = array_diff($permisos[$modulo] ?? [], (array) $acciones);
352
353            if ($restantes === []) {
354                unset($permisos[$modulo]);
355            } else {
356                $permisos[$modulo] = array_values($restantes);
357            }
358        }
359
360        // Solo módulos reconocidos: una clave inventada no concede nada.
361        return array_intersect_key($permisos, array_flip(ModuloSistema::claves()));
362    }
